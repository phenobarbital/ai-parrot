```yaml
# SDD workflow as a procedure graph (FEAT-891). Packaged asset: edit this file, then regenerate the assets.
schema_version: 1
graph: sdd
uncovered_ok: [sdd-brainstorm, sdd-brief, sdd-explain, sdd-fix, sdd-fromjira, sdd-next, sdd-proposal, sdd-status, sdd-tojira]
procedures:
  - id: spec
    title: Write the feature specification
    kind: command
    command: "/sdd-spec <slug>"
    done_when: [{probe: spec.exists}]
    next: [task]
  - id: task
    title: Decompose the spec into tasks
    kind: command
    command: "/sdd-task sdd/specs/<slug>.spec.md"
    requires_procedures: [spec]
    done_when: [{probe: index.exists}]
    next: [start]
  - id: worktree-ensure
    title: Create or reuse the feature worktree
    kind: command
    command: "parrot sdd worktree ensure --slug <slug> --feature-id <FEAT-ID>"
    requires_procedures: [task]
    preconditions: [{probe: spec.exists}, {probe: index.exists}]
    guards:
      - {kind: entry, executable: parrot, argv_prefix: [sdd, worktree, ensure], scope: {feature: {flag: "--slug"}}}
    advice: "Commit and push the spec and the task index to the base branch first; the worktree starts from the remote."
  - id: start
    title: Start a task
    kind: command
    scope_kind: task
    command: "/sdd-start <TASK>"
    requires_procedures: [task]
    done_when: [{probe: task.status, expect: "in-progress|done|done-with-issues"}]
    next: [task-close]
  - id: task-close
    title: Close a task with the script
    kind: command
    mode: mandatory
    scope_kind: task
    command: "parrot sdd task close <TASK> <slug> verified"
    use_instead: "parrot sdd task close <TASK> <slug> verified"
    requires_procedures: [start]
    preconditions:
      - {probe: task.status, expect: in-progress}
      - {probe: task.depends_done}
    done_when: [{probe: task.status, expect: "done|done-with-issues"}]
    next: [codereview]
    guards:
      - {kind: entry, executable: parrot, argv_prefix: [sdd, task, close], scope: {task: {argv: 0}, feature: {argv: 1}}}
      - {kind: bypass, executable: mv, path_globs: ["sdd/tasks/active/*"]}
      - {kind: bypass, executable: git, argv_prefix: [mv], path_globs: ["sdd/tasks/active/*"]}
    advice: "The script moves the task file and stamps the per-spec index; a hand move leaves an orphan."
  - id: codereview
    title: Review a completed task
    kind: command
    scope_kind: task
    command: "/sdd-codereview <TASK>"
    requires_procedures: [task-close]
    next: [start, done]
  - id: push
    title: Push the feature branch
    kind: command
    command: "git push -u origin <branch>"
    preconditions: [{probe: env.not_worker_sandbox}]
    done_when: [{probe: branch.pushed}]
    guards:
      - {kind: entry, executable: git, argv_prefix: [push]}
    advice: "A worker sandbox cannot reach the remote; commit and let the host run /sdd-done."
  - id: pr-create
    title: Open the pull request against the flow base branch
    kind: command
    mode: mandatory
    command: "gh pr create --base <base_branch> --head <branch>"
    preconditions: [{probe: flow.base_allowed}]
    done_when: [{probe: pr.open}]
    guards:
      - {kind: entry, executable: gh, argv_prefix: [pr, create], scope: {base: {flag: "--base|-B"}}}
    advice: "Feature and fix branches never target the hotfix base. Pass --base explicitly."
  - id: done
    title: Finish the feature
    kind: command
    command: "/sdd-done <FEAT-ID>"
    requires_procedures: [task]
    preconditions: [{probe: feature.tasks_done}]
```

### `packages/ai-parrot/tests/sdd/test_procedure_assets.py` (CREATE)
```python
"""Tests for the packaged SDD procedure manifest (FEAT-891)."""

from __future__ import annotations

import json
import shutil
from pathlib import Path

import pytest

import parrot.sdd
from parrot.knowledge.wiki.procedure.producers import load_manifest, manifest_specs
from parrot.knowledge.wiki.procedure.validate import validate_graph
from parrot.sdd.scripts.procedure_probes import PROBE_COSTS

ASSETS = Path(parrot.sdd.__file__).resolve().parent / "_assets"
PACKAGED = ASSETS / "sdd" / "procedures" / "sdd-workflow.yaml"
REPO_ROOT = Path(__file__).resolve().parents[4]


def _installed(tmp_path: Path) -> Path:
    """Copy the packaged manifest to <tmp>/sdd/procedures/ and return <tmp>."""
    target = tmp_path / "sdd" / "procedures"
    target.mkdir(parents=True)
    shutil.copy(PACKAGED, target / PACKAGED.name)
    return tmp_path


def test_sdd_manifest_valid(tmp_path: Path) -> None:
    """The packaged manifest has no finding at all; every sdd-* command is covered or listed."""
    root = _installed(tmp_path)
    specs, sources = manifest_specs(root)
    listed = set(load_manifest(sources[0], root=root).uncovered_ok)
    commands = sorted(p.stem for p in (ASSETS / "claude" / "commands").glob("sdd-*.md"))
    known = [name for name in commands if name not in listed]
    assert validate_graph(specs, known_probes=PROBE_COSTS, known_commands=known) == []
    # FILL IN: assert the nine ids, that sdd:task-close and sdd:pr-create are the only mandatory nodes,
    # and that sdd:task-close has one entry guard and two bypass guards.


def test_manifest_twin_in_sync() -> None:
    """sdd/procedures/sdd-workflow.yaml equals the packaged asset."""
    source = REPO_ROOT / "sdd" / "procedures" / "sdd-workflow.yaml"
    if not source.is_file():
        pytest.skip("not running from the monorepo checkout")
    assert source.read_bytes() == PACKAGED.read_bytes()


def test_asset_manifest_lists_the_graph() -> None:
    """The asset manifest carries the entry with the three hosts and no seed flag."""
    entries = json.loads((ASSETS / "manifest.json").read_text(encoding="utf-8"))["assets"]
    entry = next(e for e in entries if e["target"] == "sdd/procedures/sdd-workflow.yaml")
    assert entry == {"source": "sdd/procedures/sdd-workflow.yaml", "target": "sdd/procedures/sdd-workflow.yaml",
                     "hosts": ["claude", "codex", "google"]}
```
