  task closes `partial` with that recorded.
- `client.py` imports nothing from `helixdb` at module scope; the readiness probe uses
  `aiohttp` (`GET {url}/readyz`, `probe_retries` attempts, `aiohttp.ClientTimeout(total=probe_timeout_s)`,
  short backoff between attempts) — never `httpx` directly.
- Error mapping (spec §7 "Errors"): `HelixError` with `.network` / transport failure →
  `HelixUnavailable`; `.remote` with a 400-class rejection on a write → `HelixWriteRejected`
  (message carries the server code, e.g. `active_text_mutation_limit_exceeded`);
  `.embedded_unavailable` → `HelixExtraMissing` naming `ai-parrot[helixdb-embedded]`.
  `write()` on `read_only` settings raises `PermissionError` **before** touching the SDK.
- Embedded: `Disk(str(settings.root), settings.database)`; `read_only` → `embedded_reader`.
  Memory: `InMemory(settings.database)`.
- Server mode needs only `helix-db`; embedded/memory need `helixdb_uniffi` as well (check it
  with `lazy_import("helixdb_uniffi", package_name="helix-db-embedded", extra="helixdb-embedded")`
  before `AsyncClient.embedded`, so AC3 names the right extra).

### `packages/ai-parrot/src/parrot/knowledge/helix/client.py` (CREATE)
```python
"""The only HelixDB-SDK-facing module (FEAT-670 M1, G9). Async ``helixdb.AsyncClient`` only."""
from __future__ import annotations

import asyncio
import logging
from types import ModuleType
from typing import Any

import aiohttp

from parrot._imports import lazy_import  # verified: parrot/_imports.py:110
from parrot.knowledge.helix.errors import HelixExtraMissing, HelixUnavailable, HelixWriteRejected
from parrot.knowledge.helix.settings import HelixMode, HelixSettings

logger = logging.getLogger(__name__)


def _sdk() -> ModuleType:
    """Import ``helixdb`` lazily, mapping a missing install onto :class:`HelixExtraMissing`."""
    try:
        return lazy_import("helixdb", package_name="helix-db", extra="helixdb")
    except ImportError as exc:
        if isinstance(exc, HelixExtraMissing):
            raise
        raise HelixExtraMissing(str(exc)) from exc


def _require_embedded_runtime() -> None:
    """Fail with the ``helixdb-embedded`` extra name when the native runtime is absent."""
    try:
        lazy_import("helixdb_uniffi", package_name="helix-db-embedded", extra="helixdb-embedded")
    except ImportError as exc:
        raise HelixExtraMissing(str(exc)) from exc


class HelixClient:
    """Thin async wrapper over ``helixdb.AsyncClient`` with uniform error mapping."""

    def __init__(self, sdk_client: Any, settings: HelixSettings) -> None:
        self._client = sdk_client
        self.settings = settings
        self.logger = logging.getLogger(__name__)

    async def read(self, batch: Any, *, query_name: str) -> dict[str, Any]:
        """Execute a read batch; returns the ``returning`` map.

        Raises:
            HelixUnavailable: transport / server errors.
        """
        # FILL IN: build the QueryRequest from `batch` + query_name exactly as the 0.3.4 wheel expects
        #          (verify in helixdb/dsl.py / _client_common.py), await self._client.query(...),
        #          map HelixError via self._map_error(exc, write=False) — bounded by spec §7 "Errors"
        raise NotImplementedError

    async def write(self, batch: Any, *, query_name: str) -> dict[str, Any]:
        """Execute one atomic write batch.

        Raises:
            PermissionError: settings are read-only (checked before any SDK call).
            HelixWriteRejected: 400-class rejection (message carries the server code).
            HelixUnavailable: transport / server errors.
        """
        if self.settings.read_only:
            raise PermissionError(f"HelixDB database {self.settings.database!r} is opened read-only")
        # FILL IN: same request building as read(); self._client.query(...) (execute(writer_only=…) is server-only);
        #          map errors with write=True — bounded by AC11
        raise NotImplementedError

    def _map_error(self, exc: Exception, *, write: bool) -> Exception:
        """Translate a ``helixdb.HelixError`` into a FEAT-670 error."""
        # FILL IN: .network → HelixUnavailable(url/db); .remote 4xx on write → HelixWriteRejected(code);
        #          .embedded_unavailable → HelixExtraMissing("… ai-parrot[helixdb-embedded]"); else HelixUnavailable
        #          — bounded by spec §7 "map every HelixError at the client wrapper"
        raise NotImplementedError

    async def close(self) -> None:
        """Close the underlying SDK client (idempotent)."""
        if self._client is not None:
            await self._client.close()
            self._client = None

    async def __aenter__(self) -> "HelixClient":
        return self

    async def __aexit__(self, *exc: object) -> None:
        await self.close()


async def _probe_ready(settings: HelixSettings) -> None:
    """GET ``{url}/readyz`` up to ``probe_retries`` times; raise :class:`HelixUnavailable` on failure."""
    last: BaseException | None = None
    timeout = aiohttp.ClientTimeout(total=settings.probe_timeout_s)
    for attempt in range(1, settings.probe_retries + 1):
        try:
            async with aiohttp.ClientSession(timeout=timeout) as session:
                async with session.get(f"{settings.url}/readyz") as resp:
                    if resp.status == 200:
                        return
                    last = RuntimeError(f"/readyz returned HTTP {resp.status}")
        except (aiohttp.ClientError, asyncio.TimeoutError) as exc:
            last = exc
        logger.debug("HelixDB probe %s/%s failed: %s", attempt, settings.probe_retries, last)
        if attempt < settings.probe_retries:
            await asyncio.sleep(min(0.5 * attempt, 2.0))
    raise HelixUnavailable(
        f"HelixDB server at {settings.url} not ready after {settings.probe_retries} probes: {last}"
    )


async def open_helix_client(settings: HelixSettings) -> HelixClient:
    """Open a client for ``settings``.

    Server: probe ``/readyz`` then ``AsyncClient(url, api_key=…)``.
    Embedded: ``AsyncClient.embedded(Disk(root, database))`` (``embedded_reader`` when read-only).
    Memory: ``AsyncClient.embedded(InMemory(database))``.

    Raises:
        HelixExtraMissing: the required extra is not installed.
        HelixUnavailable: server not ready / embedded source refused.
    """
    sdk = _sdk()
    if settings.mode is HelixMode.SERVER:
        await _probe_ready(settings)
        return HelixClient(sdk.AsyncClient(settings.url, api_key=settings.api_key), settings)
    _require_embedded_runtime()
    # FILL IN: source = sdk.InMemory(settings.database) for MEMORY, sdk.Disk(str(settings.root), settings.database)
    #          for EMBEDDED (root.mkdir(parents=True, exist_ok=True)); opener = AsyncClient.embedded_reader if read_only
    #          else AsyncClient.embedded; wrap sdk.HelixError → HelixUnavailable (message points to server mode,
    #          refined later from the TASK-4375 spike) — bounded by AC3, AC13
    raise NotImplementedError
```
**Why this shape**: spec §3 M1 fixes `HelixClient.read/write/close` + `open_helix_client`;
`lazy_import` gives AC3 its "names the extra" message; read-only refusal is local so no
write round-trip ever happens on a reader.

import builtins

def _block_import(monkeypatch, name: str) -> None:
    real = builtins.__import__

    def fake(mod, *a, **k):
        if mod == name or mod.startswith(name + "."):
            raise ModuleNotFoundError(f"No module named {name!r}", name=name)
        return real(mod, *a, **k)

    monkeypatch.setattr(builtins, "__import__", fake)
