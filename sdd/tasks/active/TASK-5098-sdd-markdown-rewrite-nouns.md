### scratch `rewrite_sdd_verbs.py` (NOT committed — run from the worktree root with `PYTHONPATH=packages/ai-parrot/src`)
```python
import re, sys
from pathlib import Path
from parrot.sdd.cli import VERBS

ROOTS = [".claude/commands", ".claude/agents", ".claude/rules", ".agents/skills", ".agents/agents",
         ".agent/workflows", ".codex/agents", "sdd/templates", "sdd/WORKFLOW.md"]
rules: list[tuple[re.Pattern[str], str]] = []
for noun, verbs in VERBS.items():
    for verb, (module, prefix, _) in verbs.items():
        old = " ".join(["parrot sdd run", module, *prefix])
        rules.append((re.compile(rf"(?<![\w-]){re.escape(old)}(?![\w-])"), f"parrot sdd {noun} {verb}"))
rules.sort(key=lambda r: -len(r[0].pattern))  # longest (prefixed) first so `sdd_meta parse` wins over `sdd_meta`
rules.append((re.compile(r"python3? -m scripts\.sdd\.check_spec_consequences"), "parrot sdd spec consequences"))
changed = 0
for root in ROOTS:
    for path in ([Path(root)] if Path(root).is_file() else sorted(Path(root).rglob("*"))):
        if not path.is_file() or path.suffix not in {".md", ".toml", ".sh"} or "sdd-insight" in path.as_posix():
            continue
        text = new = path.read_text(encoding="utf-8")
        for pattern, repl in rules:
            new = pattern.sub(repl, new)
        if new != text:
            path.write_text(new, encoding="utf-8"); changed += 1
print(f"rewrote {changed} files", file=sys.stderr)
```
**Why this shape**: the table is the only spelling source; longest-match ordering is what keeps `parrot sdd run
sdd_meta parse` from being half-rewritten to `parrot sdd docs … parse`.
